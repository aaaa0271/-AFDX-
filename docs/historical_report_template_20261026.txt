def style(name):
    for n in (name, name.title(), name.replace(' ', '')):
        try:
            return doc.styles[n]
        except KeyError:
            continue
    return None


def clear_body():
    body = doc.element.body
    for child in list(body):
        if child.tag == qn('w:sectPr'):
            continue
        body.remove(child)


def clean(text):
    return str(text).replace('`', '')


def font(run, size=12, bold=False, cjk='宋体', ascii_font='Times New Roman', color=None, italic=False):
    run.font.name = ascii_font
    run.font.size = Pt(size)
    run.font.bold = bold
    run.font.italic = italic
    if color is not None:
        run.font.color.rgb = color
    rPr = run._element.get_or_add_rPr()
    rFonts = rPr.find(qn('w:rFonts'))
    if rFonts is None:
        rFonts = OxmlElement('w:rFonts')
        rPr.append(rFonts)
    rFonts.set(qn('w:eastAsia'), cjk)
    rFonts.set(qn('w:ascii'), ascii_font)
    rFonts.set(qn('w:hAnsi'), ascii_font)


def para(text='', indent=True, size=12, bold=False, align=None, color=None, space_before=0, space_after=0,
         line=None, left=0):
    p = doc.add_paragraph()
    pf = p.paragraph_format
    pf.line_spacing = Pt(size * 1.5)
    pf.space_before = Pt(space_before)
    pf.space_after = Pt(space_after)
    pf.left_indent = Pt(left)
    pf.first_line_indent = Pt(size * 2) if indent else Pt(0)
    if align is not None:
        p.alignment = align
    if text:
        r = p.add_run(clean(text))
        font(r, size, bold, color=color)
    return p


def heading(text, level):
    st = style('heading %d' % level) or style('Heading %d' % level)
    p = doc.add_paragraph(style=st) if st is not None else doc.add_paragraph()
    pf = p.paragraph_format
    pf.first_line_indent = Pt(0)
    pf.space_before = Pt(12 if level == 1 else 8)
    pf.space_after = Pt(6 if level == 1 else 4)
    r = p.add_run(text)
    font(r, {1: 16, 2: 14, 3: 12}[level], True, cjk='黑体')
    return p


def sub(text):
    p = doc.add_paragraph()
    pf = p.paragraph_format
    pf.first_line_indent = Pt(0)
    pf.space_before = Pt(6)
    pf.space_after = Pt(2)
    r = p.add_run(text)
    font(r, 12, True)
    return p


def bullet(text, size=12, marker='•'):
    p = doc.add_paragraph()
    pf = p.paragraph_format
    pf.left_indent = Pt(24)
    pf.first_line_indent = Pt(-12)
    pf.line_spacing = Pt(size * 1.5)
    pf.space_after = Pt(0)
    r = p.add_run(marker + ' ' + clean(text))
    font(r, size)
    return p


def mono(text):
    p = doc.add_paragraph()
    pf = p.paragraph_format
    pf.first_line_indent = Pt(0)
    pf.left_indent = Pt(12)
    pf.line_spacing = Pt(11)
    pf.space_after = Pt(6)
    r = p.add_run(text)
    font(r, 8.5, cjk='Consolas', ascii_font='Consolas')
    return p


def cell_font(cell, text, size=10.5, bold=False, align=WD_ALIGN_PARAGRAPH.LEFT):
    cell.text = ''
    p = cell.paragraphs[0]
    p.alignment = align
    p.paragraph_format.space_after = Pt(0)
    p.paragraph_format.line_spacing = Pt(size + 3)
    r = p.add_run(clean(text))
    font(r, size, bold)


def table(headers, rows, widths=None, size=10.5):
    t = doc.add_table(rows=1, cols=len(headers))
    t.alignment = WD_TABLE_ALIGNMENT.CENTER
    t.autofit = False
    tblPr = t._tbl.tblPr
    borders = OxmlElement('w:tblBorders')
    for edge in ('top', 'left', 'bottom', 'right', 'insideH', 'insideV'):
        el = OxmlElement('w:' + edge)
        el.set(qn('w:val'), 'single')
        el.set(qn('w:sz'), '4')
        el.set(qn('w:space'), '0')
        el.set(qn('w:color'), '808080')
        borders.append(el)
    tblPr.append(borders)
    layout = OxmlElement('w:tblLayout')
    layout.set(qn('w:type'), 'fixed')
    tblPr.append(layout)
    if widths:
        tblW = OxmlElement('w:tblW')
        tblW.set(qn('w:type'), 'dxa')
        tblW.set(qn('w:w'), str(int(sum(widths) * 567)))
        tblPr.append(tblW)
    # header row repeats on each page
    trPr = t.rows[0]._tr.get_or_add_trPr()
    trPr.append(OxmlElement('w:tblHeader'))
    trPr.append(OxmlElement('w:cantSplit'))
    for i, h in enumerate(headers):
        cell_font(t.rows[0].cells[i], h, size, True, WD_ALIGN_PARAGRAPH.CENTER)
        shd = OxmlElement('w:shd')
        shd.set(qn('w:fill'), 'EDF2F7')
        t.rows[0].cells[i]._tc.get_or_add_tcPr().append(shd)
    for row in rows:
        row_obj = t.add_row()
        row_obj._tr.get_or_add_trPr().append(OxmlElement('w:cantSplit'))
        cells = row_obj.cells
        for i, v in enumerate(row):
            cell_font(cells[i], str(v), size)
    if widths:
        for i, w in enumerate(widths):
            for row in t.rows:
                row.cells[i].width = Cm(w)
    p = doc.add_paragraph()
    p.paragraph_format.space_after = Pt(4)
    return t


def figure(path, width_cm=14.5, caption=''):
    p = doc.add_paragraph()
    p.alignment = WD_ALIGN_PARAGRAPH.CENTER
    p.paragraph_format.first_line_indent = Pt(0)
    p.paragraph_format.space_before = Pt(12)
    p.paragraph_format.space_after = Pt(2)
    p.add_run().add_picture(str(path), width=Cm(width_cm))
    if caption:
        c = doc.add_paragraph()
        c.alignment = WD_ALIGN_PARAGRAPH.CENTER
        c.paragraph_format.first_line_indent = Pt(0)
        c.paragraph_format.line_spacing = Pt(15)
        c.paragraph_format.space_after = Pt(6)
        r = c.add_run(clean(caption))
        font(r, 10.5, color=RGBColor(0x59, 0x59, 0x59))

    ai = RPT / "evidence/elinx_ai/status.json"
    if ai.is_file():
        d = json.loads(ai.read_text(encoding="utf-8"))
        if d.get("status") == "PASS":
            parts.append(f"AI 推理核 afdx_defense_ai_csr+TCN：{d.get('errors')} error / {d.get('warnings')} warning")
    p8 = RPT / "evidence/elinx_8port/status.json"
    if p8.is_file():
        d = json.loads(p8.read_text(encoding="utf-8"))
        if d.get("status") == "PASS":
            parts.append(f"8 端口 AFDX_top：{d.get('errors')} error / {d.get('warnings')} warning（{d.get('seconds') / 60:.1f} 分钟）")
    sec_leaves = []
    for name in ("afdx_sec_sha256", "afdx_sec_policy", "afdx_sec_time_monitor", "afdx_sec_events",
                 "afdx_sec_frame_queue", "afdx_sec_replay"):
        d = RPT / f"evidence/elinx_sec_{name.replace('afdx_sec_', '')}/status.json"
        if d.is_file():
            j = json.loads(d.read_text(encoding="utf-8"))
            if j.get("status") == "PASS":
                sec_leaves.append(name)
    if sec_leaves:
        parts.append(f"安全插件 6 个子模块（{'、'.join(n.replace('afdx_sec_', 'sec_') for n in sec_leaves)}）：均 0 error")
    st = RPT / "evidence/elinx_security/status.json"
    if st.is_file():
        d = json.loads(st.read_text(encoding="utf-8"))
        log = RPT / "evidence/elinx_security/elaboration.log"
        if d.get("status") == "PASS":
            parts.append(f"AFDX_security_plugin_top：{d.get('errors')} error / {d.get('warnings')} warning")
        elif log.is_file() and "Out of memory" in log.read_text(encoding="utf-8", errors="replace"):
            import re as _re
            mm = _re.search(r"Out of memory in module quartus_map\.exe \((\d+) megabytes used\)",
                            log.read_text(encoding="utf-8", errors="replace"))
            mb = mm.group(1) if mm else "?"
            parts.append(f"AFDX_security_plugin_top 与 AFDX_defense_top 全参数展开实测需 >{int(mb) // 1000} GB 内存"
                         f"（{mb} MB 处内存耗尽，日志见 evidence/elinx_security/ 与 evidence/elinx_defense_48g/）")
    if not parts:
        return "eLinx 分析综合按 2.2.1 的环境要求执行（原始日志见 evidence/elinx_*）。"
    return "；".join(parts) + "。以上均为 analysis/elaboration，非布局布线或时序收敛。"
